// 支持自适应检测和全息渲染的智能机器狗组件
const RobotDog = ({ className = "", status = "neutral" }) => {
  const [imgFailed, setImgFailed] = useState(false);

  // 针对白底灰色实物图的高级全息滤镜算法
  const getFilterStyle = () => {
    switch (status) {
      case 'unregistered':
        return {
          filter: 'invert(1) sepia(1) saturate(8) hue-rotate(315deg) brightness(1.1) contrast(1.3)',
        };
      case 'registered':
        return {
          filter: 'invert(1) sepia(1) saturate(6) hue-rotate(95deg) brightness(1.2) contrast(1.2)',
        };
      case 'neutral':
      default:
        return {
          filter: 'invert(1) sepia(1) saturate(5) hue-rotate(170deg) brightness(1.2) contrast(1.2)',
        };
    }
  };

  const getGlowColor = () => {
    switch (status) {
      case 'unregistered': return 'bg-red-500 shadow-[0_0_15px_#ef4444]';
      case 'registered': return 'bg-emerald-500 shadow-[0_0_15px_#10b981]';
      default: return 'bg-cyan-500 shadow-[0_0_15px_#06b6d4]';
    }
  };

  const statusColors = {
    unregistered: {
      glow: "#ef4444",
      led: "#f87171",
      accent: "#ef4444",
    },
    registered: {
      glow: "#10b981",
      led: "#34d399",
      accent: "#10b981",
    },
    neutral: {
      glow: "#06b6d4",
      led: "#22d3ee",
      accent: "#06b6d4",
    }
  }[status];

  return (
    <div className={`relative flex items-center justify-center ${className}`}>
      {/* 科技底盘感应发光圈 */}
      <div className={`absolute w-3/4 h-2 bottom-3 rounded-full blur-md opacity-45 transition-all duration-300 ${getGlowColor()}`} />
      
      {imgFailed ? (
        // 图片加载失败时，渲染精确修正了关节弯折和扁平背部的 3D 风格矢量图
        <UnitreeGo2Vector className="w-full h-full" status={status} colors={statusColors} />
      ) : (
        // 机器狗实物图片层 (过滤去除白色背景色)
        <img 
          src="image_c2f288.png" 
          alt="Unitree Go2"
          style={getFilterStyle()} 
          className="w-full h-full object-contain mix-blend-screen transition-all duration-300 drop-shadow-[0_0_12px_rgba(255,255,255,0.25)]"
          onError={() => setImgFailed(true)}
        />
      )}
    </div>
  );
};

const StatusRow = ({ label, value, status = "success", isMono = false }) => {
  const getStatusColor = () => {
    switch (status) {
      case 'success': return 'text-emerald-400';
      case 'pending': return 'text-blue-400';
      case 'warning': return 'text-yellow-400';
      case 'error': return 'text-red-400';
      default: return 'text-gray-300';
    }
  };

  return (
    <div className="flex justify-between items-center py-1.5 border-b border-blue-950/40 last:border-0 text-xs lg:text-sm">
      <span className="text-blue-200/90">{label}</span>
      <div className="flex items-center gap-1.5">
        <span className={`${getStatusColor()} ${isMono ? 'font-mono' : ''}`}>{value}</span>
        {status === 'success' && <CheckCircle2 className="w-3.5 h-3.5 text-emerald-400" />}
        {status === 'pending' && <CircleDot className="w-3.5 h-3.5 text-blue-500" />}
      </div>
    </div>

    { from: 'SRF', to: 'ACN', key: 'SRF->ACN' },
    { from: 'SRF', to: 'Computing', key: 'SRF->Computing' },
    { from: 'UPF', to: 'Computing', key: 'UPF->Computing' }
  ];

  // Dynamic loading of Three.js + GLTFLoader sequentially
  useEffect(() => {
    if (window.THREE && window.THREE.GLTFLoader) {
      setThreeLoaded(true);
      return;
    }

    const loadScript = (url) => {
      return new Promise((resolve, reject) => {
        const script = document.createElement('script');
        script.src = url;
        script.async = true;
        script.onload = resolve;
        script.onerror = reject;
        document.body.appendChild(script);
      });
    };

    loadScript('https://cdnjs.cloudflare.com/ajax/libs/three.js/r128/three.min.js')
      .then(() => loadScript('https://cdn.jsdelivr.net/npm/three@0.128.0/examples/js/loaders/GLTFLoader.js'))

      .catch(err => console.error("Error loading 3D renderer engines", err));
  }, []);

  // Handle local GLB/GLTF model upload
  const handleLocalModelUpload = (e) => {
    const file = e.target.files[0];
    if (!file) return;

    const reader = new FileReader();
    reader.onload = (event) => {
      const arrayBuffer = event.target.result;
      const THREE = window.THREE;

      if (!THREE || !THREE.GLTFLoader) return;

      const loader = new THREE.GLTFLoader();
      loader.parse(arrayBuffer, '', (gltf) => {
        const customModelGroup = gltf.scene;

        // Auto scaling and materials alignment
        customModelGroup.traverse((child) => {
          if (child.isMesh) {
            child.castShadow = true;
            child.receiveShadow = true;
            // Upgrade with reflective sci-fi look
            child.material.roughness = 0.15;
            child.material.metalness = 0.85;
          }
        });

        // Store standard scale to prevent huge GLB models breaking sand table
        const box = new THREE.Box3().setFromObject(customModelGroup);
        const size = box.getSize(new THREE.Vector3());
        const maxDim = Math.max(size.x, size.y, size.z);
        const targetSize = 1.6; // Perfect bounds size for sand table pedestal
        const scaleFactor = targetSize / maxDim;
        customModelGroup.scale.set(scaleFactor, scaleFactor, scaleFactor);
        
        // Center the imported model
        const center = box.getCenter(new THREE.Vector3());
        customModelGroup.position.set(-center.x * scaleFactor, -center.y * scaleFactor - 0.2, -center.z * scaleFactor);

        // Track and replace immediately inside the render scene
        const targetNodeGroup = nodeGroupsRef.current[selectedTargetNode];
        if (targetNodeGroup) {
          // Remove previous active children meshes except the pedestal bottom group
          const toRemove = [];
          targetNodeGroup.children.forEach(child => {
            // Keep the triple layer pedestal
            if (child.name !== 'pedestal') {
              toRemove.push(child);
            }
          });
          toRemove.forEach(child => targetNodeGroup.remove(child));

          // Mount the local uploaded GLTF/GLB model!
          targetNodeGroup.add(customModelGroup);
          setActiveSessionLogs(prev => [
            `[Local 3D Importer] Successfully loaded local GLTF model into: ${selectedTargetNode}`,
            ...prev.slice(0, 2)
          ]);
        }
      }, (error) => {
        console.error("Local GLTF parsing error:", error);
      });
    };
    reader.readAsArrayBuffer(file);
  };

  // WebGL Renderer Lifecycle
  useEffect(() => {
    if (!threeLoaded || !mountRef.current) return;

    const THREE = window.THREE;
    const container = mountRef.current;
    const width = container.clientWidth;
    const height = container.clientHeight || 450;

    scene.fog = new THREE.FogExp2(0x02050a, 0.015);

    let rotationY = 0.65;
    const radius = 26;

    const updateCameraPosition = () => {
      camera.position.x = radius * Math.sin(rotationY) * Math.cos(rotationX);
      camera.position.z = radius * Math.cos(rotationY) * Math.cos(rotationX);
      camera.position.y = radius * Math.sin(rotationX);
      camera.lookAt(0, 0, 0);
    };
    updateCameraPosition();

    // 2. Renderer
    const renderer = new THREE.WebGLRenderer({ antialias: true, alpha: true });
    renderer.setSize(width, height);
    renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));
    container.appendChild(renderer.domElement);

    // 3. Lighting
    const ambientLight = new THREE.AmbientLight(0xffffff, 0.65);
    scene.add(ambientLight);

    const dirLight1 = new THREE.DirectionalLight(0x00f0ff, 1.5);
    dirLight1.position.set(10, 20, 15);
    scene.add(dirLight1);

    const dirLight2 = new THREE.DirectionalLight(0xa855f7, 0.8);
    dirLight2.position.set(-10, -15, -10);
    scene.add(dirLight2);

    // Subtle technological ground grid
    const floorGrid = new THREE.GridHelper(32, 24, 0x1e3a8a, 0x0f172a);
    floorGrid.position.y = -1.25; 
    floorGrid.material.opacity = 0.35;
    floorGrid.material.transparent = true;
    scene.add(floorGrid);

    // Helper: Create high-fidelity double-layered Isometric Pedestal (参照图1的高科技发光基座)
    const createPedestal = (nodeColor) => {
      const pedestalGroup = new THREE.Group();
      pedestalGroup.name = 'pedestal';

      const lowerGeo = new THREE.BoxGeometry(2.4, 0.15, 2.4);

        opacity: 0.85
      });
      const lowerMesh = new THREE.Mesh(lowerGeo, lowerMat);
      lowerMesh.position.y = -1.2;
      pedestalGroup.add(lowerMesh);
